# Bump 命令片段 + 脚本骨架库（环节 6.5 Q2 使用）

按 `(ecosystem, trigger)` 交叉查表。skill-creator 在落盘阶段读取本文件对应交叉行，把「命令片段」写入派生 SKILL.md 的 `{{bumpBlock}}`，把「脚本骨架」按 `{{scriptLanguage}}` 组装成脚本本体写入 `{{scriptsDir}}/bump-version.<ext>`。

**硬约束**：

- 本文件只承载**骨架代码块**，不落盘为独立 `.cjs` / `.sh` / `.py` 文件到 release-creator 插件目录
- 骨架里的 `{{...}}` 占位符由 skill-creator 用 `{{projectAnswers}}` 字段替换后写入用户项目
- Q2 = `manual` 分支**不生成脚本**，仅写步骤

## 3. Q2 = `generate-script`（skill-creator 动态组装脚本本体落盘）

**生成脚本**到 `{{scriptsDir}}/bump-version.<ext>`。SKILL.md 里写调用行。

### 3.a `{{scriptLanguage}} = cjs`（Node.js，适合 npm / claude-plugin / 其他 JSON-based）

```javascript
#!/usr/bin/env node
/**
 * Auto-generated by release-creator (derived skill: {{targetName}})
 * Bumps version across: {{versionFiles}}
 */
const fs = require('fs');
const path = require('path');

const VERSION_FILES = {{versionFilesJSON}}; // [{ path, jsonPath }]
const VERSION_REGEX = /{{versionRegex}}/;

function getCurrent() {
  const first = VERSION_FILES[0];
  const data = JSON.parse(fs.readFileSync(first.path, 'utf8'));
  return resolveJsonPath(data, first.jsonPath);
}

function resolveJsonPath(data, p) {
  // p 形如 'version' 或 'plugins[name=token-reporter].version'
  return p.split('.').reduce((acc, seg) => {
    const m = seg.match(/^(\w+)\[(\w+)=([^\]]+)\]$/);
    if (m) return acc[m[1]].find(x => x[m[2]] === m[3]);
    return acc[seg];
  }, data);
}

function setJsonPath(data, p, value) {
  // 与 resolveJsonPath 对偶，写回
  const parts = p.split('.');
  const last = parts.pop();
  const parent = parts.reduce((acc, seg) => {
    const m = seg.match(/^(\w+)\[(\w+)=([^\]]+)\]$/);
    if (m) return acc[m[1]].find(x => x[m[2]] === m[3]);
    return acc[seg];
  }, data);
  parent[last] = value;
}

function bump(cur, type) {
  {{#if versionStructure === 'semver'}}
  const [a, b, c] = cur.split('.').map(Number);
  if (type === 'major') return `${a + 1}.0.0`;
  if (type === 'minor') return `${a}.${b + 1}.0`;
  if (type === 'patch') return `${a}.${b}.${c + 1}`;
  {{/if}}
  {{#if versionStructure === 'calver-ym-serial'}}
  const ym = new Date().toISOString().slice(2, 7).replace('-', '.');
  const [curYm, serial] = [cur.split('.').slice(0, 2).join('.'), cur.split('.')[2]];
  return curYm === ym ? `${ym}.${Number(serial) + 1}` : `${ym}.0`;
  {{/if}}
  if (type && VERSION_REGEX.test(type)) return type; // 直接指定版本
  throw new Error(`Unknown bump type: ${type}`);
}

function main() {
  const cur = getCurrent();
  const next = bump(cur, process.argv[2] || 'patch');
  if (!VERSION_REGEX.test(next)) throw new Error(`Invalid version: ${next}`);
  for (const f of VERSION_FILES) {
    const data = JSON.parse(fs.readFileSync(f.path, 'utf8'));
    setJsonPath(data, f.jsonPath, next);
    fs.writeFileSync(f.path, JSON.stringify(data, null, 2) + '\n');
    console.log(`✅ ${f.path} → ${next}`);
  }
  console.log(next);
}

main();
```

### 3.c `{{scriptLanguage}} = sh`（Bash，适合 docs-only / 无 JSON/TOML 的场景）

```bash
#!/usr/bin/env bash
# Auto-generated by release-creator (derived skill: {{targetName}})
set -euo pipefail

VERSION_FILE="{{versionFiles[0].path}}"
VERSION_REGEX='{{versionRegex}}'

cur=$(grep -oE "$VERSION_REGEX" "$VERSION_FILE" | head -1)
type="${1:-patch}"

case "$type" in
  major|minor|patch)
    IFS=. read -r a b c <<< "$cur"
    case "$type" in
      major) nxt="$((a+1)).0.0" ;;
      minor) nxt="$a.$((b+1)).0" ;;
      patch) nxt="$a.$b.$((c+1))" ;;
    esac
    ;;
  *)
    if [[ "$type" =~ $VERSION_REGEX ]]; then nxt="$type"; else echo "bad bump"; exit 1; fi
    ;;
esac

sed -i.bak "s/$cur/$nxt/" "$VERSION_FILE"
rm -f "$VERSION_FILE.bak"
echo "$nxt"
```

## 组装规则（skill-creator 调用）

1. 读 `{{projectAnswers}}.bumpTrigger`（对应 6.5 Q2 的值）
2. 匹配本文件对应章节
3. 把 `{{bumpBlock}}` 片段写入派生 SKILL.md
4. 若章节标注「生成脚本」→ 按 `{{scriptLanguage}}` 选对应子节，组装脚本本体写入 `{{scriptsDir}}/bump-version.{{scriptExt}}`
5. 若对应生态/语言组合标注 `TODO:` → 脚本落盘时改写为 `TODO: <说明>` 注释块，不写可执行代码
6. 脚本文件落盘后 `chmod +x`（由 skill-creator 处理，本骨架不包含 chmod 指令）
